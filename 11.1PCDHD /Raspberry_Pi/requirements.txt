// 2. HELPER FUNCTIONS
// ============================================================

// Safely update an existing element using plain text.
function setText(id, value) {

    const element = document.getElementById(id);

    if (element) {
        element.textContent = value;
    }
}


// Reject null, undefined, NaN and infinite values; zero remains valid.
function validNumber(value) {

    return typeof value === "number" &&
           Number.isFinite(value);
}


// Format valid measurements to one decimal place with optional units.
function format(value, unit = "") {

    if (!validNumber(value)) {
        return "Unavailable";
    }

    return (
        value.toFixed(1) +
        (unit ? " " + unit : "")
    );
}


// Show a numeric measurement or an explicit pending message.
function displayMeasurement(id, value, unit, message) {

    const element = document.getElementById(id);

    if (validNumber(value)) {

        element.textContent = format(value, unit);
        element.classList.remove("pending");

    } else {

        element.textContent = message;
        element.classList.add("pending");
    }
}


// ============================================================
// 3. CLEAR INVALID OR DISCONNECTED DATA
// ============================================================

// Clear live fields when data is missing or disconnected; keep the reference input.
function clearBed(bed, status) {

    setText("status" + bed, status);

    document.getElementById("status" + bed).className =
        "status warning";

    document.getElementById("liquid" + bed).style.height =
        "0%";

    setText("percent" + bed, "--");
    setText("volume" + bed, "-- mL");

    const fields = [

        "weight",
        "drops",
        "dpm",
        "flow",
        "dropflow",
        "difference",
        "comparison",
        "sensor",
        "bottle",
        "updated"

    ];

    fields.forEach(field => {

        const element =
            document.getElementById(field + bed);

        element.textContent = "--";
        element.classList.remove("pending");

    });
}


// ============================================================
// 4. UPDATE SYSTEM STATUS
// ============================================================

// Translate status codes into readable labels and color classes.
function updateStatus(bed, statusValue) {

    const status =
        document.getElementById("status" + bed);

    status.textContent =
        String(statusValue).replaceAll("_", " ");

    if (statusValue === "NORMAL") {

        status.className = "status normal";

    } else if (

        statusValue === "LOW_VOLUME" ||
        statusValue === "DISTURBANCE" ||
        statusValue === "INITIALIZING"

    ) {

        status.className = "status warning";

    } else {

        status.className = "status danger";

    }
}


// ============================================================
// 5. UPDATE INDIVIDUAL BED
// ============================================================

// Update only the card corresponding to the received bed identifier.
function updateBed(b) {

    const bed = b.bed;

    // Do not display stale or missing readings as live values.
    if (!b.available) {

        clearBed(bed, b.status);
        return;

    }

    updateStatus(bed, b.status);


    // --------------------------------------------------------
    // Remaining volume and percentage
    // --------------------------------------------------------

    const percentage = b.percentage;

    const liquid =
        document.getElementById("liquid" + bed);

    // Clamp the visual fill level to the range 0–100 percent.
    liquid.style.height = validNumber(percentage)

        ? Math.max(0, Math.min(100, percentage)) + "%"

        : "0%";

    // Use amber for a low remaining percentage, otherwise blue.
    liquid.style.background =

        validNumber(percentage) && percentage <= 20

        ? "#f59e0b"

        : "#38bdf8";

    setText(
        "percent" + bed,
        format(percentage, "%")
    );

    setText(
        "volume" + bed,
        format(b.volume_ml, "mL")
    );


    // --------------------------------------------------------
    // Load cell measurements
    // --------------------------------------------------------

    setText(
        "weight" + bed,
        format(b.total_weight_g, "g")
    );


    // --------------------------------------------------------
    // Optical drop measurements
    // --------------------------------------------------------

    setText(
        "drops" + bed,
        validNumber(b.drop_count)
            ? b.drop_count
            : "--"
    );

    displayMeasurement(
        "dpm" + bed,
        b.drops_per_minute,
        "drops/min",
        "Awaiting measurement"
    );


    // --------------------------------------------------------
    // Weight-based flow
    // --------------------------------------------------------

    displayMeasurement(
        "flow" + bed,
        b.weight_flow_mlh,
        "mL/h",
        "Awaiting valid measurement"
    );


    // --------------------------------------------------------
    // Drop-based flow
    // --------------------------------------------------------

    displayMeasurement(
        "dropflow" + bed,
        b.drop_flow_mlh,
        "mL/h",
        "Not calibrated / unavailable"
    );


    // --------------------------------------------------------
    // Compare both measurement methods
    // --------------------------------------------------------

    displayMeasurement(
        "difference" + bed,
        b.flow_difference_mlh,
        "mL/h",
        "Awaiting both measurements"
    );


    // --------------------------------------------------------
    // Demonstration reference comparison
    // --------------------------------------------------------

    const target =
        document.getElementById("target" + bed);

    // Convert the optional demonstration reference to a number.
    const targetValue = Number(target.value);

    if (

        target.value !== "" &&
        Number.isFinite(targetValue) &&
        targetValue >= 0 &&
        validNumber(b.weight_flow_mlh)

    ) {

        // Calculate the signed difference between measured weight flow and the reference.
        const difference =
            b.weight_flow_mlh - targetValue;

        setText(
            "comparison" + bed,
            difference.toFixed(1) + " mL/h"
        );

    } else {

        setText(
            "comparison" + bed,
            "Awaiting reference / measurement"
        );

    }


    // --------------------------------------------------------
    // Sensor status
    // --------------------------------------------------------

    setText(
        "sensor" + bed,
        b.sensor_ok ? "WORKING" : "FAULT"
    );

    setText(
        "bottle" + bed,
        b.bottle_present ? "ATTACHED" : "REMOVED"
    );

    setText(
        "updated" + bed,
        b.last_updated || "--"
    );

}


// ============================================================
// 6. FETCH LIVE SENSOR DATA
// ============================================================

// Request current readings from the Flask backend without reloading the page.
async function updateDashboard() {

    try {

        // Disable caching so the latest server response is requested.
        const response = await fetch("/api/data", {

            cache: "no-store"

        });

        if (!response.ok) {

            throw new Error("Server unavailable");

        }

        const data = await response.json();

        const connection =
            document.getElementById("connection");

        // Update both cards when Bluetooth is connected; otherwise clear live data.
        if (data.connected) {

            connection.textContent =
                "Bluetooth Connected";

            connection.className =
                "connection normal";

            data.beds.forEach(updateBed);

        } else {

            connection.textContent =
                "Bluetooth Disconnected";

            connection.className =
                "connection danger";

            clearBed(1, "CONNECTION LOST");
            clearBed(2, "CONNECTION LOST");

        }

    } catch (error) {

        setText(
            "connection",
            "Dashboard connection lost"
        );

        document.getElementById("connection").className =
            "connection danger";

        clearBed(1, "CONNECTION LOST");
        clearBed(2, "CONNECTION LOST");

    }
}


// ============================================================
// 7. DRAW REMAINING VOLUME HISTORY
// ============================================================

// Draw the recent volume history for one bed using the Canvas API.
function drawChart(bed, readings) {

    const canvas =
        document.getElementById("chart" + bed);

    const ctx = canvas.getContext("2d");

    const w = canvas.width;
    const h = canvas.height;

    // Erase the previous graph before drawing a new one.
    ctx.clearRect(0, 0, w, h);

    ctx.font = "12px Arial";
    ctx.fillStyle = "#94a3b8";

    ctx.fillText("500 mL", 5, 15);
    ctx.fillText("250 mL", 5, h / 2);
    ctx.fillText("0 mL", 5, h - 5);

    ctx.strokeStyle = "#334155";
    ctx.lineWidth = 1;

    // Draw five horizontal reference lines.
    for (let i = 0; i <= 4; i++) {

        const y = 20 + i * (h - 35) / 4;

        ctx.beginPath();
        ctx.moveTo(55, y);
        ctx.lineTo(w - 10, y);
        ctx.stroke();

    }

    ctx.strokeStyle = "#38bdf8";
    ctx.lineWidth = 2;

    ctx.beginPath();

    let started = false;

    // Plot samples in their stored order, not by elapsed time.
    readings.forEach((reading, index) => {

        // Break the line when a measurement is unavailable.
        if (!validNumber(reading.volume)) {

            started = false;
            return;

        }

        const x = 55 +

            index /
            Math.max(1, readings.length - 1) *
            (w - 65);

        // Clamp the plotted volume to the graph range of 0–500 mL.
        const value = Math.max(
            0,
            Math.min(500, reading.volume)
        );

        const y =

            h - 15 -
            value / 500 * (h - 35);

        if (!started) {

            ctx.moveTo(x, y);
            started = true;

        } else {

            ctx.lineTo(x, y);

        }

    });

    ctx.stroke();

}


// ============================================================
// 8. FETCH MEASUREMENT HISTORY
// ============================================================

// Request the recent history for both beds from the Flask API.
async function updateHistory() {

    try {

        const response = await fetch("/api/history", {

            cache: "no-store"

        });

        if (!response.ok) {
            throw new Error("History unavailable");
        }

        const history = await response.json();

        drawChart(1, history["1"] || []);
        drawChart(2, history["2"] || []);

    } catch (error) {

        drawChart(1, []);
        drawChart(2, []);

    }

}


// ============================================================
// 9. AUTOMATIC DASHBOARD REFRESH
// ============================================================

// Fetch live data and history sequentially, then schedule the next refresh.
async function refresh() {

    await updateDashboard();

    await updateHistory();

    // Wait one second after both requests finish before repeating.
    setTimeout(refresh, 1000);

}

// Start the automatic refresh cycle.
refresh();
